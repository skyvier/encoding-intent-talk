  \note{Luottamusraja ei tarkoita pelkästään verkon reunaa. Se syntyy aina,
  kun ohjelma alkaa tehdä arvosta uuden oletuksen.

  HTTP-pyynnön sisältöön ei vielä luoteta. Validoinnin jälkeen voimme luottaa
  tiettyihin rakenteellisiin ominaisuuksiin. Allekirjoitettuun JWT:hen ei
  luoteta ennen sen todentamista. Todennettu identiteetti ei puolestaan vielä
  tarkoita, että käyttäjä saa käsitellä juuri tätä resurssia.

  Ajonaikaisia tarkistuksia tarvitaan edelleen. Tyyppijärjestelmän tehtävä ei
  ole korvata niitä, vaan säilyttää tieto siitä, mitä on jo tarkistettu.}

  \note{Ilman tyyppitason erottelua myöhempi funktio saa helposti tavallisen
  merkkijonon, tunnisteen tai boolean-arvon. Sen täytyy silloin luottaa siihen,
  että joku muu on jo tehnyt tarvittavan tarkistuksen. Todistetta vaativa
  rajapinta tekee tarkistuksen ohittamisesta näkyvän. Todisteen pitää liittyä
  juuri siihen käyttäjään, organisaatioon, resurssiin ja toimintoon, jota
  käsitellään.

  Yksinkertaisessa sovelluksessa näin raskas mekanismi voi olla tarpeeton.
  Mutta jos tarkistuksen unohtaminen voi aiheuttaa tietovuodon, väärän asiakkaan
  tietojen käsittelyn tai peruuttamattoman liiketoimintatoiminnon, koneiston
  kustannus voi olla hyvin perusteltu.

  Tavoitteena ei ole tehdä kaikesta todistettavaa. Tavoitteena on tunnistaa
  rajat, joiden vahingossa ohittaminen olisi erityisen kallista.}

\begin{frame}[fragile]{Johda tyypistä väärinkäyttötestin vaativat kentät}

type StringLikeKey = {
  [K in keyof AiSearchFilters]-?:
    NonNullable<AiSearchFilters[K]> extends string | string[]
      ? K : never;
}[keyof AiSearchFilters];

// "query" | "city" | "postalCode"
\end{lstlisting}
  \bottomline{Uusi merkkijonokenttä muuttaa unionia automaattisesti.}
\end{frame}

\begin{frame}[fragile]{Kattavan testitaulukon tyyppi}
\begin{lstlisting}[style=deckcode,language=TypeScript]
type OverlongValue<K extends StringLikeKey> =
  NonNullable<AiSearchFilters[K]> extends string[]
    ? string[] : string;

type TestCases = {
  [K in StringLikeKey]: {
    maxLength: number;
    validValue: OverlongValue<K>;
    makeValue: (text: string) => OverlongValue<K>;
  };
};
\end{lstlisting}
  \bottomline{Uudesta merkkijonokentästä tulee pakollinen testitapaus.}
\end{frame}

\begin{frame}[fragile]{Täydellinen testitaulukko}
\begin{lstlisting}[style=deckcode,language=TypeScript,basicstyle=\ttfamily\footnotesize\color{DeckText}]
const cases = {
  query: {
    maxLength: 120, validValue: ["ok"],
    makeValue: (s) => [s],
  },
  city: {
    maxLength: 120, validValue: ["ok"],
    makeValue: (s) => [s],
  },
  postalCode: {
    maxLength: 10, validValue: "00100",
    makeValue: (s) => s,
  },
} satisfies TestCases;
\end{lstlisting}
  \bottomline{Jätä \texttt{postalCode} pois, niin TypeScript osoittaa puuttuvan tapauksen.}
\end{frame}

\begin{frame}[fragile]{Testaa seuraavaksi ajonaikainen raja}
